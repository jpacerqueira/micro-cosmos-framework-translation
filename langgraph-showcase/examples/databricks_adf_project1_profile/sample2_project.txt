# src/etl1.py
from utils.data_profiler import DataProfiler
from pyspark.sql import SparkSession

class ETL1:
    def __init__(self):
        self.spark = SparkSession.builder.appName("ETL1Job").getOrCreate()

    def run(self, input_path, output_path, table_name):
        df = self.spark.read.format("delta").load(input_path)
        profiler = DataProfiler()
        expected_schema = profiler.get_expected_schema(table_name)
        profiler.validate_schema(df, expected_schema)
        profiler.profile(df)
        df_clean = df.dropna()
        df_clean.write.format("delta").mode("overwrite").save(output_path)

# src/utils/data_profiler.py
from pyspark.sql.types import StructType, StructField, StringType, IntegerType

class DataProfiler:
    def get_expected_schema(self, table_name):
        if table_name == "sample_table":
            return StructType([
                StructField("id", IntegerType(), True),
                StructField("val1", StringType(), True),
                StructField("val2", StringType(), True),
                StructField("val3", StringType(), True),
                StructField("val4", StringType(), True)
            ])
        raise ValueError(f"Unknown table name: {table_name}")

    def validate_schema(self, df, expected_schema):
        if df.schema != expected_schema:
            raise ValueError("Schema validation failed.")

    def profile(self, df):
        print(f"Column Count: {len(df.columns)}")
        df.printSchema()
        df.describe().show()

import unittest

# Azure Databricks & Azure Functions ETL Project

## Structure
- `src/` contains PySpark ETL code.
- `utils/` has helper classes like data profiling and schema validation.
- `tests/` has unit tests.
